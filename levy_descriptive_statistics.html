Property tax levies and tax bases in Cook County, IL

This page compares the source taxing-agency records before sample exclusions with the grouped panel used in the corrected manuscript analysis after exclusions. Within every comparison, the after-exclusion results appear first and are therefore the default tab.

The annual-change variables follow the corrected model definitions:

  • d_levy is 100 times the log change in the total final levy.
  • d_eav is 100 times the log change in equalized assessed value (EAV).

The before-exclusion results use the original PTAXSIM agency number as the unit. The after-exclusion results use the grouped taxing agency used in the manuscript. The two tabs therefore show how both filtering and agency grouping change the sample.

Sample composition

The tabs compare the same 2008–2023 period. The before-exclusion tab counts original PTAXSIM agency numbers; the after-exclusion tab counts grouped manuscript agencies.

    group_by(agency_id) |>
    mutate(
      d_levy = 100 * log(total_final_levy / lag(total_final_levy)),
      d_eav = 100 * log(eav / lag(eav))
    ) |>
    ungroup()

  before_change_df <- before_df |> filter(year > first_level_year)
  before_available <- nrow(before_df) > 0
}

before_missing_message <- paste0(
  "The before-exclusion source file was not found. Add agency_raw_joined.csv ",
  "or raw_agency_joined.csv to the project root to display this tab."
)

make_overview <- function(df, change_df, id_var, panel_label) {
  years <- sort(unique(df$year))
  panel_year_count <- length(years)
  id_sym <- rlang::ensym(id_var)

  years_per_agency <- df |>
    distinct(!!id_sym, year) |>
    count(!!id_sym, name = "years_observed")

  tibble(
    Measure = c(
      "Panel definition",
      "First year",
      "Last year",
      "Years in panel",
      "Agency-year observations",
      "Unique taxing-agency units",
      "Units observed in every panel year",
      "Agency-year observations with a prior year",
      "Observations with finite d_levy and d_eav"
    ),
    Value = c(
      panel_label,
      min(years),
      max(years),
      panel_year_count,
      nrow(df),
      n_distinct(df[[rlang::as_name(id_sym)]]),
      sum(years_per_agency$years_observed == panel_year_count),
      nrow(change_df),
      sum(is.finite(change_df$d_levy) & is.finite(change_df$d_eav))
    )
  )
}

make_long_run_growth <- function(df, id_var) {
  id_sym <- rlang::ensym(id_var)

  start_values <- df |>
    filter(year == first_level_year) |>
    transmute(
      agency_id = as.character(!!id_sym),
      agency_name_display,
      agency_type,
      start_levy = total_final_levy,
      start_eav = eav
    )

  end_values <- df |>
    filter(year == last_year) |>
    transmute(
      agency_id = as.character(!!id_sym),
      end_levy = total_final_levy,
      end_eav = eav
    )

  start_values |>
    inner_join(end_values, by = "agency_id") |>
    mutate(
      levy_growth = 100 * (end_levy - start_levy) / start_levy,
      eav_growth = 100 * (end_eav - start_eav) / start_eav,
      levy_minus_eav_growth = levy_growth - eav_growth
    ) |>
    filter(
      is.finite(levy_growth),
      is.finite(eav_growth)
    )
}

long_run_growth_after <- make_long_run_growth(main_df, agency_id)
long_run_growth_before <- if (before_available) {
  make_long_run_growth(before_df, agency_id)
} else {
  NULL
}
